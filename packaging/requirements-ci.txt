# CI dependency pins for the Windows release build (v1.0.4+).
#
# EXACT versions on purpose: the release pipeline must build with
# the same libraries every time, never with whatever happens to be
# newest on release day. Every version below exists on PyPI
# (verified 2026-10-19). Bump deliberately, in review, never by
# loosening a pin.
#
# The Argus package itself is installed separately with
# `pip install --no-deps ".[gui,windows]"` so these pins stay the
# single source of truth (the pyproject extras are minimums for
# end users installing from source, not for the release build).
#
# P1-27: EVERY build-relevant package is pinned here — not just
# the four headline names. PyInstaller's own dependency tree
# (hooks-contrib, pefile, altgraph, pywin32-ctypes, packaging,
# setuptools) decides what the frozen exe contains and how its
# analysis runs; leaving any of them floating means the release
# build is not reproducible even with PyInstaller itself pinned.

pywin32==312
pyinstaller==6.22.3
pyinstaller-hooks-contrib==2026.8
pefile==2024.8.26
altgraph==0.17.5
